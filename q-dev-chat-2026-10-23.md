This is a two-microcontroller Arduino robot project called **Buddy Robot**. Here's the breakdown:

## Purpose
A companion/buddy robot with a face-like display that reacts to its environment using computer vision and sensors.

## Architecture
Two ESP32 boards communicating over UART Serial:

- **Master ESP32** (`Buddy_Robot.ino`) — the main brain:
  - Drives an ST7735 TFT display showing robot status (changes color based on mood)
  - Reads ultrasonic distance sensor (HC-SR04)
  - Controls motors via L298N driver
  - Controls RGB LED and buzzer
  - Listens for mood signals from the CAM module

- **ESP32-CAM** (`CAM.ino`) — the vision module:
  - Captures images and connects to Wi-Fi
  - Sends emotion signals (`"HAPPY"` / `"SAD"`) to the Master over UART
  - Uses the Gemini AI API key (likely for image-based emotion/scene analysis)

- **tests** (`tests.ino`) — utility sketch:
  - I2C scanner to detect connected devices (MPU6050 gyro is wired but not yet in main code)

## Hardware
| Component | Role |
|---|---|
| ST7735 TFT | Robot "face" display |
| ESP32-CAM | Vision + Wi-Fi + Gemini AI |
| MPU6050 | Gyroscope (wired, not yet coded) |
| L298N | Motor driver for movement |
| HC-SR04 | Obstacle detection |
| RGB LED + Buzzer | Expressive feedback |

## Notable Issues (from pinout audit)
- GPIO 0 used as CAM TX — boot conflict risk if CAM is connected during reset
- HC-SR04 Echo outputs 5V but ESP32 GPIO 12 is 3.3V tolerant — needs a voltage divider
- Pin conflicts in `Buddy_Robot.ino`: RGB pins defined as 25/26/27 but the pinout table assigns 26/27 to L298N motors and 25/32/33 to RGB

#include &lt;Adafruit_GFX.h&gt;
#include &lt;Adafruit_ST7735.h&gt;
#include &lt;SPI.h&gt;

#define TFT_CS     10
#define TFT_RST     8
#define TFT_DC      9

// Initialize 1.44&quot; ST7735 display (128x128)
Adafruit_ST7735 tft = Adafruit_ST7735(TFT_CS, TFT_DC, TFT_RST);

// Eye configuration settings (Global variables so all functions can see them)
int eyeW = 24;
int eyeH = 36;
int eyeR = 6;
int eyeG = 8;

// Function prototypes
void drawEyes(int shiftX, int shiftY, int hLeft, int hRight, int eyebrowAngle);
void blink();

void setup() {
  // Initialize 1.44&quot; 128x128 ST7735 display chip
  tft.initR(INITR_144GREENTAB); 
  
  tft.setRotation(0); // Set landscape orientation
  tft.fillScreen(ST7735_BLACK);
}

void loop() {
  // Center &amp; Look around
  drawEyes(0, 0, eyeH, eyeH, 0);
  delay(1000);
  
  blink();
  
  drawEyes(10, 0, eyeH, eyeH, 0);  // Look Right
  delay(800);
  
  drawEyes(-10, 0, eyeH, eyeH, 0); // Look Left
  delay(800);
  
  drawEyes(0, 0, eyeH, eyeH, 0);
  delay(500);

  // Angry Expression
  drawEyes(0, 4, eyeH, eyeH, 12);
  delay(1200);

  // Curious Expression
  drawEyes(0, 0, eyeH + 4, 16, -8);
  delay(1200);

  // Wink Expression
  drawEyes(0, 0, eyeH, 0, 0);
  delay(500);
  
  blink();
}

// --- Helper Drawing Functions ---

void drawEyes(int shiftX, int shiftY, int hLeft, int hRight, int eyebrowAngle) {
  tft.fillScreen(ST7735_BLACK);

  int curCenterX = (tft.width() / 2) + shiftX;
  int curCenterY = (tft.height() / 2) + shiftY;

  int leftX  = curCenterX - eyeW - (eyeG / 2);
  int leftY  = curCenterY - (hLeft / 2);
  
  int rightX = curCenterX + (eyeG / 2);
  int rightY = curCenterY - (hRight / 2);

  // Draw Left Eye
  if (hLeft &gt; 2) {
    tft.fillRoundRect(leftX, leftY, eyeW, hLeft, eyeR, ST7735_CYAN);
  } else {
    tft.drawFastHLine(leftX, curCenterY, eyeW, ST7735_CYAN);
  }

  // Draw Right Eye
  if (hRight &gt; 2) {
    tft.fillRoundRect(rightX, rightY, eyeW, hRight, eyeR, ST7735_CYAN);
  } else {
    tft.drawFastHLine(rightX, curCenterY, eyeW, ST7735_CYAN);
  }

  // Eyebrow cutouts for mood expressions
  if (eyebrowAngle &gt; 0) {
    tft.fillTriangle(leftX - 2, leftY - 2, leftX + eyeW + 2, leftY - 2, leftX + eyeW + 2, leftY + eyebrowAngle, ST7735_BLACK);
    tft.fillTriangle(rightX - 2, rightY - 2, rightX + eyeW + 2, rightY - 2, rightX - 2, rightY + eyebrowAngle, ST7735_BLACK);
  } else if (eyebrowAngle &lt; 0) {
    int d = -eyebrowAngle;
    tft.fillTriangle(leftX - 2, leftY - 2, leftX + eyeW + 2, leftY - 2, leftX - 2, leftY + d, ST7735_BLACK);
    tft.fillTriangle(rightX - 2, rightY - 2, rightX + eyeW + 2, rightY - 2, rightX + eyeW + 2, rightY + d, ST7735_BLACK);
  }
}

void blink() {
  for (int h = eyeH; h &gt;= 0; h -= 8) {
    drawEyes(0, 0, h, h, 0);
    delay(10);
  }
  for (int h = 0; h &lt;= eyeH; h += 8) {
    drawEyes(0, 0, h, h, 0);
    delay(10);
  }
}
